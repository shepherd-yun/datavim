# sqlvim v0.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付 sqlvim v0.1：读取配置连接 MySQL/PostgreSQL，在 lazygit 式多面板 TUI 中浏览连接/库/schema/表树，分页查看表数据（横向滚动、`K` 查看单元格），查看表结构；带 `?` 帮助和 which-key 风格的 leader 提示。

**Architecture:** Go + Bubble Tea v2（Elm 架构）。`internal/db` 定义唯一的数据库接缝（`Driver`/`Conn`/`MetaReader`/`Dialect`），MySQL 与 PG 各自实现并跑同一套合约测试。`internal/session` 管理密码（Keychain）和按 (连接, 库) 复用的连接。UI 层由纯逻辑组件（keymap、tree、grid、structure、overlay）和一个根 Model 组成；所有 IO 都在 `tea.Cmd` 中异步执行。

**Tech Stack:** Go 1.26 · charm.land/bubbletea/v2 · charm.land/lipgloss/v2 · github.com/charmbracelet/x/ansi · github.com/jackc/pgx/v5 · github.com/go-sql-driver/mysql · github.com/zalando/go-keyring · github.com/pelletier/go-toml/v2 · github.com/google/go-cmp · github.com/testcontainers/testcontainers-go (+ modules/mysql, modules/postgres)

**Spec:** `docs/superpowers/specs/2026-10-13-sqlvim-design.md`

## Global Constraints

- Go module 路径：`github.com/shepherd-yun/sqlvim`；`go.mod` 的 go 指令为 `1.26`。
- 只支持 `mysql` 和 `postgres` 两种驱动。
- 配置目录：`$XDG_CONFIG_HOME/sqlvim`，未设置时为 `~/.config/sqlvim`（**不要**用 `os.UserConfigDir`，它在 macOS 上返回 `~/Library/Application Support`）。
- 状态目录：`$XDG_STATE_HOME/sqlvim`，未设置时为 `~/.local/state/sqlvim`；日志写到 `sqlvim.log`，权限 `0600`。
- 配置文件里不能出现密码；Keychain 的 service 为 `sqlvim`，account 为连接名。
- 日志和错误信息中不能出现密码。
- 连接超时 10 秒；元数据加载超时 15 秒；表数据查询超时 30 秒。
- 默认 `page_size = 200`，`max_column_width = 40`。
- 默认 leader 键为 `space`。
- 所有 UI 文案使用英文（README 中英双语）。
- UI 层不得直接做 IO；所有数据库调用都在 `tea.Cmd` 中执行。
- 提交信息遵循 Conventional Commits，并以对话中给定的 attribution 行结尾。
- 集成测试使用 build tag `integration`，需要 Docker（本机为 OrbStack，先运行 `orb start`）。

### v0.1 明确不做（推迟到后续版本）

`<Space>n` 新建连接表单（v0.2）、列表内 `/` 搜索（v0.2）、`zh`/`zl` 横向滚动（光标跟随已足够，v0.2 再评估）、readonly 强制（v0.2）、SSH 隧道（v0.3，配置了 `ssh` 的连接在 v0.1 会返回明确错误）、断线自动重连（v0.2）、生产库红框（v0.3，v0.1 只在树里标 `[PROD]`）。

## Review Focus

1. **单元格里有换行、制表符、ANSI 转义序列或 CJK 宽字符** → 表格保持对齐，终端不被注入转义序列（Task 8 `textutil` 测试、Task 12 grid 测试）。
2. **密码含有 `@ : / ? # %` 等 URL/DSN 特殊字符** → 能正确连接（Task 5 `connString` 测试、Task 6 `dsnConfig` 测试）。
3. **需要引号的标识符**（保留字表名 `order`、大小写混合 `MixedCase`、内嵌引号） → 浏览和结构查询都正常（Task 4 `Quote` 测试、Task 5/6 合约测试）。
4. **主机不可达、密码错误、服务未启动** → 错误显示在树节点上，程序不退出，修复后再按 `l` 能重试（Task 7 "失败不缓存"测试、Task 15 错误节点测试）。
5. **终端极小（如 10×3、0×0）或缩放到很小** → 渲染不 panic（Task 12、Task 15 极小尺寸测试）。

```go
// Command sqlvim is a terminal SQL client with vim keys.
package main

import (
	"flag"
	"fmt"
	"io"
	"os"
)

// version is set at build time via -ldflags "-X main.version=...".
var version = "dev"

func main() {
	if err := run(os.Args[1:], os.Stdout); err != nil {
		fmt.Fprintln(os.Stderr, "sqlvim:", err)
		os.Exit(1)
	}
}

func run(args []string, stdout io.Writer) error {
	fs := flag.NewFlagSet("sqlvim", flag.ContinueOnError)
	showVersion := fs.Bool("version", false, "print version and exit")
	if err := fs.Parse(args); err != nil {
		return err
	}
	if *showVersion {
		fmt.Fprintln(stdout, "sqlvim", version)
		return nil
	}
	return nil
}
```

```go
// Package config loads sqlvim's TOML configuration files.
package config

import (
	"bytes"
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"

	"github.com/pelletier/go-toml/v2"
)

// Env classifies a connection; prod enables safety features.
type Env string

const (
	EnvDev     Env = "dev"
	EnvStaging Env = "staging"
	EnvProd    Env = "prod"
)

// Connection is one entry of connections.toml. Passwords are never stored here.
type Connection struct {
	Name     string            `toml:"name"`
	Driver   string            `toml:"driver"`
	Host     string            `toml:"host"`
	Port     int               `toml:"port"`
	User     string            `toml:"user"`
	Database string            `toml:"database"`
	Env      Env               `toml:"env"`
	ReadOnly *bool             `toml:"readonly"`
	SSH      string            `toml:"ssh"`
	Params   map[string]string `toml:"params"`
}

// IsReadOnly reports the effective read-only flag: explicit value, else true for prod.
func (c Connection) IsReadOnly() bool {
	if c.ReadOnly != nil {
		return *c.ReadOnly
	}
	return c.Env == EnvProd
}

// Settings is config.toml.
type Settings struct {
	PageSize       int `toml:"page_size"`
	MaxColumnWidth int `toml:"max_column_width"`
}

// DefaultSettings returns the settings used when config.toml omits a key.
func DefaultSettings() Settings {
	return Settings{PageSize: 200, MaxColumnWidth: 40}
}

// Config is everything loaded from the config directory.
type Config struct {
	Connections []Connection
	Settings    Settings
}

var defaultPorts = map[string]int{"mysql": 3306, "postgres": 5432}

type connectionsFile struct {
	Connections []Connection `toml:"connection"`
}

// Load reads connections.toml and config.toml from dir. Missing files are not an error.
func Load(dir string) (*Config, error) {
	cfg := &Config{Settings: DefaultSettings()}

	connPath := filepath.Join(dir, "connections.toml")
	var cf connectionsFile
	if err := decodeFile(connPath, &cf); err != nil {
		return nil, err
	}
	conns, err := normalize(cf.Connections)
	if err != nil {
		return nil, fmt.Errorf("%s: %w", connPath, err)
	}
	cfg.Connections = conns

	settingsPath := filepath.Join(dir, "config.toml")
	if err := decodeFile(settingsPath, &cfg.Settings); err != nil {
		return nil, err
	}
	if cfg.Settings.PageSize <= 0 {
		return nil, fmt.Errorf("%s: page_size must be positive", settingsPath)
	}
	if cfg.Settings.MaxColumnWidth <= 0 {
		return nil, fmt.Errorf("%s: max_column_width must be positive", settingsPath)
	}
	return cfg, nil
}

func decodeFile(path string, v any) error {
	data, err := os.ReadFile(path)
	if errors.Is(err, fs.ErrNotExist) {
		return nil
	}
	if err != nil {
		return err
	}
	dec := toml.NewDecoder(bytes.NewReader(data))
	dec.DisallowUnknownFields()
	if err := dec.Decode(v); err != nil {
		var de *toml.DecodeError
		if errors.As(err, &de) {
			row, col := de.Position()
			return fmt.Errorf("%s:%d:%d: %s", path, row, col, de.Error())
		}
		var se *toml.StrictMissingError
		if errors.As(err, &se) {
			return fmt.Errorf("%s: %s", path, se.String())
		}
		return fmt.Errorf("%s: %w", path, err)
	}
	return nil
}

func normalize(in []Connection) ([]Connection, error) {
	seen := make(map[string]bool, len(in))
	out := make([]Connection, 0, len(in))
	for i, c := range in {
		if c.Name == "" {
			return nil, fmt.Errorf("connection #%d: name is required", i+1)
		}
		if seen[c.Name] {
			return nil, fmt.Errorf("connection %q: duplicate name", c.Name)
		}
		seen[c.Name] = true
		port, ok := defaultPorts[c.Driver]
		if !ok {
			return nil, fmt.Errorf("connection %q: unknown driver %q (want mysql or postgres)", c.Name, c.Driver)
		}
		if c.Port == 0 {
			c.Port = port
		}
		if c.Host == "" {
			c.Host = "localhost"
		}
		switch c.Env {
		case "":
			c.Env = EnvDev
		case EnvDev, EnvStaging, EnvProd:
		default:
			return nil, fmt.Errorf("connection %q: unknown env %q (want dev, staging or prod)", c.Name, c.Env)
		}
		out = append(out, c)
	}
	return out, nil
}

// Dir returns $XDG_CONFIG_HOME/sqlvim or ~/.config/sqlvim.
func Dir() (string, error) {
	return xdgDir("XDG_CONFIG_HOME", ".config")
}

// StateDir returns $XDG_STATE_HOME/sqlvim or ~/.local/state/sqlvim.
func StateDir() (string, error) {
	return xdgDir("XDG_STATE_HOME", filepath.Join(".local", "state"))
}

func xdgDir(env, fallback string) (string, error) {
	if base := os.Getenv(env); base != "" {
		return filepath.Join(base, "sqlvim"), nil
	}
	home, err := os.UserHomeDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(home, fallback, "sqlvim"), nil
}
```

```go
// Package secret stores connection passwords in the OS keychain.
package secret

import (
	"errors"
	"os"
	"strings"

	"github.com/zalando/go-keyring"
)

const service = "sqlvim"

// ErrNotFound means no password is stored for the connection.
var ErrNotFound = errors.New("secret not found")

// Store reads and writes connection passwords.
type Store interface {
	Get(name string) (string, error)
	Set(name, password string) error
}

// Keyring is a Store backed by the macOS Keychain (or the platform equivalent).
// The environment variable EnvVar(name) takes precedence, which is handy on CI.
type Keyring struct {
	// Getenv defaults to os.Getenv.
	Getenv func(string) string
}

// Get returns the password for the named connection.
func (k Keyring) Get(name string) (string, error) {
	getenv := k.Getenv
	if getenv == nil {
		getenv = os.Getenv
	}
	if v := getenv(EnvVar(name)); v != "" {
		return v, nil
	}
	pw, err := keyring.Get(service, name)
	if errors.Is(err, keyring.ErrNotFound) {
		return "", ErrNotFound
	}
	return pw, err
}

// Set stores the password for the named connection.
func (Keyring) Set(name, password string) error {
	return keyring.Set(service, name, password)
}

// EnvVar is the environment variable that overrides the keychain, e.g.
// "prod-pg" -> "SQLVIM_PASSWORD_PROD_PG".
func EnvVar(name string) string {
	return "SQLVIM_PASSWORD_" + strings.Map(func(r rune) rune {
		switch {
		case r >= 'a' && r <= 'z':
			return r - 'a' + 'A'
		case r >= 'A' && r <= 'Z', r >= '0' && r <= '9':
			return r
		default:
			return '_'
		}
	}, name)
}
```

```go
import (

// Quote wraps an identifier in q, doubling any embedded q.
func Quote(s string, q rune) string {
	qs := string(q)
	return qs + strings.ReplaceAll(s, qs, qs+qs) + qs
}

// LimitOffset appends a LIMIT/OFFSET clause; MySQL and PostgreSQL share this syntax.
func LimitOffset(sql string, limit, offset int) string {
	return fmt.Sprintf("%s LIMIT %d OFFSET %d", sql, limit, offset)
```

```go
package db

import (
	"context"
	"errors"
)

// ErrNotFound is returned when a requested object does not exist.
var ErrNotFound = errors.New("not found")

// ConnConfig is what a driver needs to open a connection.
type ConnConfig struct {
	Host     string
	Port     int
	User     string
	Password string
	Database string
	Params   map[string]string
}

// Driver opens connections for one database engine.
type Driver interface {
	Name() string
	// DatabaseScoped reports whether a Conn can only see the database it was
	// opened with (true for PostgreSQL, false for MySQL).
	DatabaseScoped() bool
	Open(ctx context.Context, cfg ConnConfig) (Conn, error)
}

// Conn is an open, pooled connection. It is safe for concurrent use.
type Conn interface {
	Query(ctx context.Context, sql string, args ...any) (*ResultSet, error)
	Exec(ctx context.Context, sql string, args ...any) (ExecResult, error)
	Meta() MetaReader
	Dialect() Dialect
	Close() error
}

// MetaReader reads catalog information. Per-table methods return empty
// results for a missing table, except DDL which returns ErrNotFound.
type MetaReader interface {
	Databases(ctx context.Context) ([]string, error)
	// Schemas returns nil for engines without schemas (MySQL).
	Schemas(ctx context.Context, database string) ([]string, error)
	Tables(ctx context.Context, ns Namespace) ([]Table, error)
	Columns(ctx context.Context, t TableRef) ([]Column, error)
	Indexes(ctx context.Context, t TableRef) ([]Index, error)
	ForeignKeys(ctx context.Context, t TableRef) ([]ForeignKey, error)
	DDL(ctx context.Context, t TableRef) (string, error)
}

// Dialect hides SQL syntax differences between engines.
type Dialect interface {
	QuoteIdent(s string) string
	QualifiedName(t TableRef) string
	Paginate(sql string, limit, offset int) string
}
```

```go
// Package postgres implements db.Driver for PostgreSQL using pgx.
package postgres

import (
	"context"
	"fmt"
	"net"
	"net/url"
	"strconv"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"

	"github.com/shepherd-yun/sqlvim/internal/db"
)

// Driver opens PostgreSQL connections.
type Driver struct{}

func (Driver) Name() string         { return "postgres" }
func (Driver) DatabaseScoped() bool { return true }

// Open connects and pings the server.
func (Driver) Open(ctx context.Context, cfg db.ConnConfig) (db.Conn, error) {
	pcfg, err := pgxpool.ParseConfig(connString(cfg))
	if err != nil {
		return nil, err
	}
	pcfg.MaxConns = 4
	pool, err := pgxpool.NewWithConfig(ctx, pcfg)
	if err != nil {
		return nil, err
	}
	if err := pool.Ping(ctx); err != nil {
		pool.Close()
		return nil, err
	}
	return &conn{pool: pool, database: pcfg.ConnConfig.Database}, nil
}

func connString(cfg db.ConnConfig) string {
	database := cfg.Database
	if database == "" {
		database = "postgres"
	}
	q := url.Values{}
	q.Set("connect_timeout", "10")
	for k, v := range cfg.Params {
		q.Set(k, v)
	}
	u := url.URL{
		Scheme:   "postgres",
		User:     url.UserPassword(cfg.User, cfg.Password),
		Host:     net.JoinHostPort(cfg.Host, strconv.Itoa(cfg.Port)),
		Path:     "/" + database,
		RawQuery: q.Encode(),
	}
	return u.String()
}

type conn struct {
	pool     *pgxpool.Pool
	database string
}

func (c *conn) Dialect() db.Dialect   { return dialect{} }
func (c *conn) Meta() db.MetaReader   { return meta{c: c} }
func (c *conn) Close() error          { c.pool.Close(); return nil }

func (c *conn) Exec(ctx context.Context, sql string, args ...any) (db.ExecResult, error) {
	tag, err := c.pool.Exec(ctx, sql, args...)
	if err != nil {
		return db.ExecResult{}, err
	}
	return db.ExecResult{RowsAffected: tag.RowsAffected()}, nil
}

// Query asks the server for text-format results so every value is shown
// exactly as psql would print it.
func (c *conn) Query(ctx context.Context, sql string, args ...any) (*db.ResultSet, error) {
	rows, err := c.pool.Query(ctx, sql, append([]any{pgx.QueryResultFormats{pgx.TextFormatCode}}, args...)...)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	typeMap := rows.Conn().TypeMap()
	fields := rows.FieldDescriptions()
	rs := &db.ResultSet{Columns: make([]db.ColumnInfo, len(fields))}
	for i, f := range fields {
		typ := fmt.Sprintf("oid:%d", f.DataTypeOID)
		if t, ok := typeMap.TypeForOID(f.DataTypeOID); ok {
			typ = t.Name
		}
		rs.Columns[i] = db.ColumnInfo{Name: f.Name, DatabaseType: typ}
	}
	for rows.Next() {
		raw := rows.RawValues()
		row := make([]db.Value, len(raw))
		for i, b := range raw {
			if b == nil {
				row[i] = db.Value{Null: true}
			} else {
				row[i] = db.Value{Text: string(b)}
			}
		}
		rs.Rows = append(rs.Rows, row)
	}
	return rs, rows.Err()
}

type dialect struct{}

func (dialect) QuoteIdent(s string) string { return db.Quote(s, '"') }

func (d dialect) QualifiedName(t db.TableRef) string {
	if t.Schema == "" {
		return d.QuoteIdent(t.Name)
	}
	return d.QuoteIdent(t.Schema) + "." + d.QuoteIdent(t.Name)
}

func (dialect) Paginate(sql string, limit, offset int) string {
	return db.LimitOffset(sql, limit, offset)
}
```

```go
	def  string

// buildTableDDL assembles CREATE TABLE text; PostgreSQL has no SHOW CREATE TABLE.
func buildTableDDL(qualified string, cols []db.Column, cons []constraint, indexDefs []string) string {
	q := dialect{}
	lines := make([]string, 0, len(cols)+len(cons))
	for _, c := range cols {
		l := "    " + q.QuoteIdent(c.Name) + " " + c.Type
		if !c.Nullable {
			l += " NOT NULL"
		}
		if c.Default != nil {
			l += " DEFAULT " + *c.Default
		}
		lines = append(lines, l)
	}
	for _, c := range cons {
		lines = append(lines, "    CONSTRAINT "+q.QuoteIdent(c.name)+" "+c.def)
	}
	var b strings.Builder
	fmt.Fprintf(&b, "CREATE TABLE %s (\n%s\n);\n", qualified, strings.Join(lines, ",\n"))
	for _, d := range indexDefs {
		b.WriteString(d + ";\n")
	}
	return b.String()
}
```

```go
	}

func regclass(t db.TableRef) string { return dialect{}.QualifiedName(t) }

func (m meta) Databases(ctx context.Context) ([]string, error) {

		WHERE NOT datistemplate AND datallowconn ORDER BY 1`)
}

func (m meta) Schemas(ctx context.Context, database string) ([]string, error) {
	if err := m.checkDatabase(database); err != nil {
		return nil, err
	}
	return m.strings(ctx, `SELECT nspname::text FROM pg_namespace
		WHERE nspname !~ '^pg_' AND nspname <> 'information_schema' ORDER BY 1`)

func (m meta) Tables(ctx context.Context, ns db.Namespace) ([]db.Table, error) {
	if err := m.checkDatabase(ns.Database); err != nil {
		return nil, err
	}
	rows, err := m.c.pool.Query(ctx, `
		SELECT c.relname::text, c.relkind::text, COALESCE(obj_description(c.oid, 'pg_class'), '')
		FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace
		WHERE n.nspname = $1 AND c.relkind IN ('r','p','v','m','f') AND NOT c.relispartition
		ORDER BY 1`, ns.Schema)
	if err != nil {
		return nil, err
	}
	return pgx.CollectRows(rows, func(row pgx.CollectableRow) (db.Table, error) {
		var t db.Table
		var kind string
		err := row.Scan(&t.Name, &kind, &t.Comment)
		if kind == "v" || kind == "m" {
			t.Kind = db.TableKindView
		}
		return t, err
	})
}

func (m meta) Columns(ctx context.Context, t db.TableRef) ([]db.Column, error) {
	rows, err := m.c.pool.Query(ctx, `
		SELECT a.attname::text,
		       format_type(a.atttypid, a.atttypmod),
		       NOT a.attnotnull,
		       pg_get_expr(d.adbin, d.adrelid),
		       COALESCE(col_description(a.attrelid, a.attnum), ''),
		       EXISTS (SELECT 1 FROM pg_index i
		               WHERE i.indrelid = a.attrelid AND i.indisprimary AND a.attnum = ANY(i.indkey))
		FROM pg_attribute a
		LEFT JOIN pg_attrdef d ON d.adrelid = a.attrelid AND d.adnum = a.attnum
		WHERE a.attrelid = to_regclass($1) AND a.attnum > 0 AND NOT a.attisdropped
		ORDER BY a.attnum`, regclass(t))
	if err != nil {
		return nil, err
	}
	return pgx.CollectRows(rows, func(row pgx.CollectableRow) (db.Column, error) {
		var c db.Column
		err := row.Scan(&c.Name, &c.Type, &c.Nullable, &c.Default, &c.Comment, &c.PrimaryKey)
		return c, err
	})
}

	rows, err := m.c.pool.Query(ctx, `
		SELECT i.relname::text, ix.indisunique, ix.indisprimary,
		       ARRAY(SELECT pg_get_indexdef(ix.indexrelid, k, true)
		             FROM generate_series(1, ix.indnkeyatts) AS k ORDER BY k)
		FROM pg_index ix JOIN pg_class i ON i.oid = ix.indexrelid
		WHERE ix.indrelid = to_regclass($1)
		ORDER BY 1`, regclass(t))
	if err != nil {
		return nil, err
	}
	return pgx.CollectRows(rows, func(row pgx.CollectableRow) (db.Index, error) {
		var i db.Index
		err := row.Scan(&i.Name, &i.Unique, &i.Primary, &i.Columns)
		return i, err
	})
}

func (m meta) ForeignKeys(ctx context.Context, t db.TableRef) ([]db.ForeignKey, error) {
	rows, err := m.c.pool.Query(ctx, `
		SELECT con.conname::text,
		       ARRAY(SELECT a.attname::text FROM unnest(con.conkey) WITH ORDINALITY AS k(attnum, ord)
		             JOIN pg_attribute a ON a.attrelid = con.conrelid AND a.attnum = k.attnum ORDER BY k.ord),
		       rn.nspname::text, rc.relname::text,
		       ARRAY(SELECT a.attname::text FROM unnest(con.confkey) WITH ORDINALITY AS k(attnum, ord)
		             JOIN pg_attribute a ON a.attrelid = con.confrelid AND a.attnum = k.attnum ORDER BY k.ord)
		FROM pg_constraint con
		JOIN pg_class rc ON rc.oid = con.confrelid
		JOIN pg_namespace rn ON rn.oid = rc.relnamespace
		WHERE con.contype = 'f' AND con.conrelid = to_regclass($1)
		ORDER BY 1`, regclass(t))
	if err != nil {
		return nil, err
	}
	return pgx.CollectRows(rows, func(row pgx.CollectableRow) (db.ForeignKey, error) {
		var fk db.ForeignKey
		var schema, table string
		err := row.Scan(&fk.Name, &fk.Columns, &schema, &table, &fk.RefColumns)
		fk.RefTable = db.TableRef{Namespace: db.Namespace{Database: m.c.database, Schema: schema}, Name: table}
		return fk, err
	})
}

func (m meta) DDL(ctx context.Context, t db.TableRef) (string, error) {
	name := regclass(t)
	var kind string
	err := m.c.pool.QueryRow(ctx, `SELECT relkind::text FROM pg_class WHERE oid = to_regclass($1)`, name).Scan(&kind)
	if errors.Is(err, pgx.ErrNoRows) {
		return "", fmt.Errorf("%s: %w", name, db.ErrNotFound)
	}
	if err != nil {
		return "", err
	}
	if kind == "v" || kind == "m" {
		var def string
		if err := m.c.pool.QueryRow(ctx, `SELECT pg_get_viewdef(to_regclass($1), true)`, name).Scan(&def); err != nil {
			return "", err
		}
		keyword := "VIEW"
		if kind == "m" {
			keyword = "MATERIALIZED VIEW"
		}
		return fmt.Sprintf("CREATE %s %s AS\n%s", keyword, name, def), nil
	}

	cols, err := m.Columns(ctx, t)
	if err != nil {
		return "", err
	}
	rows, err := m.c.pool.Query(ctx, `
		SELECT conname::text, pg_get_constraintdef(oid, true) FROM pg_constraint
		WHERE conrelid = to_regclass($1) AND contype IN ('p','u','f','c','x')
		ORDER BY contype <> 'p', conname`, name)
	if err != nil {
		return "", err
	}
	cons, err := pgx.CollectRows(rows, func(row pgx.CollectableRow) (constraint, error) {
		var c constraint
		err := row.Scan(&c.name, &c.def)
		return c, err
	})
	if err != nil {
		return "", err
	}
	// Indexes that back a constraint are already covered by the constraint.
	indexDefs, err := m.strings(ctx, `
		SELECT pg_get_indexdef(indexrelid) FROM pg_index
		WHERE indrelid = to_regclass($1)
		  AND indexrelid NOT IN (SELECT conindid FROM pg_constraint WHERE conrelid = to_regclass($1))
		ORDER BY 1`, name)
	if err != nil {
		return "", err
	}
	return buildTableDDL(name, cols, cons, indexDefs), nil
}
```

```go
// Driver opens MySQL connections.
type Driver struct{}

func (Driver) Name() string         { return "mysql" }
func (Driver) DatabaseScoped() bool { return false }

// Open connects and pings the server.
func (Driver) Open(ctx context.Context, cfg db.ConnConfig) (db.Conn, error) {
	mc, err := dsnConfig(cfg)
	if err != nil {
		return nil, err
	}
	connector, err := gomysql.NewConnector(mc)
	if err != nil {
		return nil, err
	}
	sqldb := sql.OpenDB(connector)
	sqldb.SetMaxOpenConns(4)
	if err := sqldb.PingContext(ctx); err != nil {
		_ = sqldb.Close()
		return nil, err
	}
	return &conn{db: sqldb}, nil
}

func dsnConfig(cfg db.ConnConfig) (*gomysql.Config, error) {
	mc := gomysql.NewConfig()
	mc.User = cfg.User
	mc.Passwd = cfg.Password
	mc.Net = "tcp"
	mc.Addr = net.JoinHostPort(cfg.Host, strconv.Itoa(cfg.Port))
	mc.DBName = cfg.Database
	mc.InterpolateParams = true
	mc.Timeout = 10 * time.Second
	if len(cfg.Params) == 0 {
		return mc, nil
	}
	// Round-trip through the DSN so special keys (tls, charset, ...) are parsed
	// by the driver exactly as in a hand-written DSN.
	q := url.Values{}
	for k, v := range cfg.Params {
		q.Set(k, v)
	}
	dsn := mc.FormatDSN()
	sep := "?"
	if strings.Contains(dsn[strings.LastIndex(dsn, "/"):], "?") {
		sep = "&"
	}
	return gomysql.ParseDSN(dsn + sep + q.Encode())
}

	rows, err := c.db.QueryContext(ctx, query, args...)
	if err != nil {
		return nil, err
	}

		return nil, err

		rs.Columns[i] = db.ColumnInfo{Name: ct.Name(), DatabaseType: ct.DatabaseTypeName()}

	ptrs := make([]any, len(types))

		ptrs[i] = &dest[i]

			return nil, err

		for i, v := range dest {

				row[i] = db.Value{Text: displayText(v.String)}
			} else {
				row[i] = db.Value{Null: true}
			}
		}
		rs.Rows = append(rs.Rows, row)
	}
```

```go
	}
	defer rows.Close()
	var out []db.Table
	for rows.Next() {
		var t db.Table
		var typ string
		var comment sql.NullString
		if err := rows.Scan(&t.Name, &typ, &comment); err != nil {
			return nil, err
		}
		if strings.Contains(typ, "VIEW") {
			t.Kind = db.TableKindView
		} else {
			t.Comment = comment.String
		}
		out = append(out, t)
	}
	return out, rows.Err()
}

func (m meta) Columns(ctx context.Context, t db.TableRef) ([]db.Column, error) {
	rows, err := m.c.db.QueryContext(ctx, `
		SELECT COLUMN_NAME, COLUMN_TYPE, IS_NULLABLE, COLUMN_DEFAULT, COLUMN_COMMENT, COLUMN_KEY
		FROM information_schema.COLUMNS
		WHERE TABLE_SCHEMA = ? AND TABLE_NAME = ? ORDER BY ORDINAL_POSITION`, t.Database, t.Name)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []db.Column
	for rows.Next() {
		var c db.Column
		var nullable, key string
		var def sql.NullString
		if err := rows.Scan(&c.Name, &c.Type, &nullable, &def, &c.Comment, &key); err != nil {
			return nil, err
		}
		c.Nullable = nullable == "YES"
		c.PrimaryKey = key == "PRI"
		if def.Valid {
			c.Default = &def.String
		}
		out = append(out, c)
	}
	return out, rows.Err()
}

func (m meta) Indexes(ctx context.Context, t db.TableRef) ([]db.Index, error) {
	rows, err := m.c.db.QueryContext(ctx, `
		SELECT INDEX_NAME, NON_UNIQUE, COALESCE(COLUMN_NAME, '(expression)')
		FROM information_schema.STATISTICS
		WHERE TABLE_SCHEMA = ? AND TABLE_NAME = ? ORDER BY INDEX_NAME, SEQ_IN_INDEX`, t.Database, t.Name)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []db.Index
	for rows.Next() {
		var name, col string
		var nonUnique int
		if err := rows.Scan(&name, &nonUnique, &col); err != nil {
			return nil, err
		}
		if n := len(out); n > 0 && out[n-1].Name == name {
			out[n-1].Columns = append(out[n-1].Columns, col)
			continue
		}
		out = append(out, db.Index{Name: name, Columns: []string{col}, Unique: nonUnique == 0, Primary: name == "PRIMARY"})
	}
	return out, rows.Err()
}

func (m meta) ForeignKeys(ctx context.Context, t db.TableRef) ([]db.ForeignKey, error) {
	rows, err := m.c.db.QueryContext(ctx, `
		SELECT CONSTRAINT_NAME, COLUMN_NAME, REFERENCED_TABLE_SCHEMA, REFERENCED_TABLE_NAME, REFERENCED_COLUMN_NAME
		FROM information_schema.KEY_COLUMN_USAGE
		WHERE TABLE_SCHEMA = ? AND TABLE_NAME = ? AND REFERENCED_TABLE_NAME IS NOT NULL
		ORDER BY CONSTRAINT_NAME, ORDINAL_POSITION`, t.Database, t.Name)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []db.ForeignKey
	for rows.Next() {
		var name, col, refDB, refTable, refCol string
		if err := rows.Scan(&name, &col, &refDB, &refTable, &refCol); err != nil {
			return nil, err
		}
		if n := len(out); n > 0 && out[n-1].Name == name {
			out[n-1].Columns = append(out[n-1].Columns, col)
			out[n-1].RefColumns = append(out[n-1].RefColumns, refCol)
			continue
		}
		out = append(out, db.ForeignKey{
			Name: name, Columns: []string{col},
			RefTable:   db.TableRef{Namespace: db.Namespace{Database: refDB}, Name: refTable},
			RefColumns: []string{refCol},
		})
	}
	return out, rows.Err()
}

func (m meta) DDL(ctx context.Context, t db.TableRef) (string, error) {
	name := dialect{}.QualifiedName(t)
	notFound := fmt.Errorf("%s: %w", name, db.ErrNotFound)
	rows, err := m.c.db.QueryContext(ctx, "SHOW CREATE TABLE "+name)
	if err != nil {
		var me *gomysql.MySQLError
		if errors.As(err, &me) && me.Number == errNoSuchTable {
			return "", notFound
		}
		return "", err
	}
	defer rows.Close()
	cols, err := rows.Columns()
	if err != nil {
		return "", err
	}
	if !rows.Next() {
		return "", notFound
	}
	// Tables return (Table, Create Table); views return (View, Create View, ...).
	vals := make([]sql.NullString, len(cols))
	ptrs := make([]any, len(cols))
	for i := range vals {
		ptrs[i] = &vals[i]
	}
	if err := rows.Scan(ptrs...); err != nil {
		return "", err
	}
	return vals[1].String, nil
}
```

```go
// Package session owns passwords and open connections for configured servers.
package session

import (
	"context"
	"errors"
	"fmt"
	"sync"

	"github.com/shepherd-yun/sqlvim/internal/config"
	"github.com/shepherd-yun/sqlvim/internal/db"
	"github.com/shepherd-yun/sqlvim/internal/secret"
)

// ErrPasswordRequired means no password is known; ask the user.
var ErrPasswordRequired = errors.New("password required")

type connKey struct {
	conn     string
	database string
}

// Manager opens connections lazily and reuses them. Safe for concurrent use.
type Manager struct {
	conns   []config.Connection
	drivers map[string]db.Driver
	secrets secret.Store

	mu        sync.Mutex
	passwords map[string]string
	open      map[connKey]db.Conn
}

// NewManager builds a Manager for the configured connections.
func NewManager(conns []config.Connection, secrets secret.Store, drivers ...db.Driver) *Manager {
	m := &Manager{
		conns:     conns,
		drivers:   make(map[string]db.Driver, len(drivers)),
		secrets:   secrets,
		passwords: map[string]string{},
		open:      map[connKey]db.Conn{},
	}
	for _, d := range drivers {
		m.drivers[d.Name()] = d
	}
	return m
}

// Connections returns the configured connections in file order.
func (m *Manager) Connections() []config.Connection { return m.conns }

// Conn returns an open connection to database ("" means the configured default).
func (m *Manager) Conn(ctx context.Context, name, database string) (db.Conn, error) {
	cfg, drv, err := m.lookup(name)
	if err != nil {
		return nil, err
	}
	if cfg.SSH != "" {
		return nil, fmt.Errorf("connection %q: ssh tunnels are not supported yet", name)
	}
	if database == "" {
		database = cfg.Database
	}
	key := connKey{conn: name}
	if drv.DatabaseScoped() {
		key.database = database
	}

	m.mu.Lock()
	c, ok := m.open[key]
	m.mu.Unlock()
	if ok {
		return c, nil
	}

	pw, err := m.password(name)
	if err != nil {
		return nil, err
	}
	c, err = drv.Open(ctx, db.ConnConfig{
		Host: cfg.Host, Port: cfg.Port, User: cfg.User, Password: pw,
		Database: database, Params: cfg.Params,
	})
	if err != nil {
		return nil, fmt.Errorf("connect %s: %w", name, err)
	}

	m.mu.Lock()
	defer m.mu.Unlock()
	if existing, ok := m.open[key]; ok {
		_ = c.Close() // lost a race with another opener
		return existing, nil
	}
	m.open[key] = c
	return c, nil
}

// SetPassword remembers a password for this run and optionally saves it to
// the keychain. Open connections for name are closed so the next use reconnects.
func (m *Manager) SetPassword(name, password string, save bool) error {
	m.mu.Lock()
	m.passwords[name] = password
	for k, c := range m.open {
		if k.conn == name {
			_ = c.Close()
			delete(m.open, k)
		}
	}
	m.mu.Unlock()
	if save {
		return m.secrets.Set(name, password)
	}
	return nil
}

// CloseAll closes every open connection.
func (m *Manager) CloseAll() {
	m.mu.Lock()
	defer m.mu.Unlock()
	for k, c := range m.open {
		_ = c.Close()
		delete(m.open, k)
	}
}

func (m *Manager) lookup(name string) (config.Connection, db.Driver, error) {
	for _, c := range m.conns {
		if c.Name == name {
			drv, ok := m.drivers[c.Driver]
			if !ok {
				return c, nil, fmt.Errorf("connection %q: driver %q not available", name, c.Driver)
			}
			return c, drv, nil
		}
	}
	return config.Connection{}, nil, fmt.Errorf("unknown connection %q", name)
}

func (m *Manager) password(name string) (string, error) {
	m.mu.Lock()
	pw, ok := m.passwords[name]
	m.mu.Unlock()
	if ok {
		return pw, nil
	}
	pw, err := m.secrets.Get(name)
	if errors.Is(err, secret.ErrNotFound) {
		return "", ErrPasswordRequired
	}
	if err != nil {
		return "", fmt.Errorf("read password for %s: %w", name, err)
	}
	m.mu.Lock()
	m.passwords[name] = pw
	m.mu.Unlock()
	return pw, nil
}
```

```go
// Package textutil holds display-width aware string helpers for the TUI.
package textutil

import (
	"strings"

	"github.com/charmbracelet/x/ansi"
)

// Sanitize makes database text safe to draw on one terminal line.
func Sanitize(s string) string {
	return strings.Map(func(r rune) rune {
		switch {
		case r == '\n':
			return '↵'
		case r == '\t':
			return ' '
		case r == '\r':
			return -1
		case r < 0x20, r == 0x7f, r >= 0x80 && r < 0xa0:
			return '�'
		}
		return r
	}, s)
}

// Fit truncates s to width cells (ending in "…") and pads it to exactly width.
func Fit(s string, width int) string {
	if width <= 0 {
		return ""
	}
	s = ansi.Truncate(s, width, "…")
	if pad := width - ansi.StringWidth(s); pad > 0 {
		s += strings.Repeat(" ", pad)
	}
	return s
}

// PadLines returns s with exactly n lines.
func PadLines(s string, n int) string {
	var lines []string
	if s != "" {
		lines = strings.Split(s, "\n")
	}
	for len(lines) < n {
		lines = append(lines, "")
	}
	return strings.Join(lines[:n], "\n")
}

// Wrap hard-wraps s into lines of at most width cells.
func Wrap(s string, width int) []string {
	if width <= 0 {
		return nil
	}
	var lines []string
	var cur strings.Builder
	w := 0
	for _, r := range s {
		rw := ansi.StringWidth(string(r))
		if w+rw > width && w > 0 {
			lines = append(lines, cur.String())
			cur.Reset()
			w = 0
		}
		cur.WriteRune(r)
		w += rw
	}
	return append(lines, cur.String())
}

// Table lays out rows under headers with two-space gaps.
func Table(headers []string, rows [][]string) []string {
	widths := make([]int, len(headers))
	for i, h := range headers {
		widths[i] = ansi.StringWidth(h)
	}
	for _, r := range rows {
		for i, c := range r {
			widths[i] = max(widths[i], ansi.StringWidth(Sanitize(c)))
		}
	}
	line := func(cells []string) string {
		parts := make([]string, len(cells))
		for i, c := range cells {
			parts[i] = Fit(Sanitize(c), widths[i])
		}
		return strings.TrimRight(strings.Join(parts, "  "), " ")
	}
	rule := make([]string, len(widths))
	for i, w := range widths {
		rule[i] = strings.Repeat("─", w)
	}
	out := []string{line(headers), strings.TrimRight(strings.Join(rule, "  "), " ")}
	if len(rows) == 0 {
		return append(out, "(none)")
	}
	for _, r := range rows {
		out = append(out, line(r))
	}
	return out
}

// Clamp limits v to [lo, hi]; if hi < lo it returns lo.
func Clamp(v, lo, hi int) int {
	if hi < lo || v < lo {
		return lo
	}
	if v > hi {
		return hi
	}
	return v
}
```

```go
		return ""

		head = ansi.Truncate(" "+title+" ", inner, "…")
	}

	lines := make([]string, 0, height)
	lines = append(lines, border.Render("┌")+Title.Render(head)+border.Render(strings.Repeat("─", rest)+"┐"))

	for i := range height - 2 {
		l := ""
		if i < len(bodyLines) {
			l = bodyLines[i]
		}
		lines = append(lines, border.Render("│")+textutil.Fit(l, inner)+border.Render("│"))
	}
	lines = append(lines, border.Render("└"+strings.Repeat("─", inner)+"┘"))
	return strings.Join(lines, "\n")
```

```go
	NextSection   Action = "section.next"

const tabPrefix = "tab.goto."

// GotoTab is the action that focuses tab n (1-based).
func GotoTab(n int) Action { return Action(tabPrefix + strconv.Itoa(n)) }

// TabNumber extracts n from a GotoTab action.
func TabNumber(a Action) (int, bool) {
	s, ok := strings.CutPrefix(string(a), tabPrefix)
	if !ok {
		return 0, false
	}
	n, err := strconv.Atoi(s)
	return n, err == nil
}

// LeaderToken in Binding.Keys is replaced by the configured leader key.
const LeaderToken = "<leader>"

// Binding maps a key sequence to an action within a context.
type Binding struct {
	Context Context
	Keys    []string
	Action  Action
	Desc    string
}

// Result is the outcome of resolving a key sequence.
type Result int

const (
	NoMatch Result = iota
	Pending
	Matched
)

// Hint is one which-key line: the remaining keys and what they do.
type Hint struct {
	Keys string
	Desc string
}

// HelpEntry is one help line; aliases for the same action are merged.
type HelpEntry struct {
	Context Context
	Keys    string
	Desc    string
}

// Keymap resolves key sequences.
type Keymap struct {
	leader   string
	bindings []Binding
}

// New expands the leader token and validates bindings.
func New(leader string, bindings []Binding) (*Keymap, error) {
	k := &Keymap{leader: leader}
	for _, b := range bindings {
		keys := make([]string, len(b.Keys))
		for i, key := range b.Keys {
			if key == LeaderToken {
				key = leader
			}
			keys[i] = key
		}
		b.Keys = keys
		k.bindings = append(k.bindings, b)
	}
	if err := k.validate(); err != nil {
		return nil, err
	}
	return k, nil
}

// Default returns the built-in keymap with <space> as leader.
func Default() *Keymap {
	k, err := New("space", DefaultBindings())
	if err != nil {
		panic(err) // covered by TestDefaultIsValid
	}
	return k
}

// Leader returns the leader key.
func (k *Keymap) Leader() string { return k.leader }

func (k *Keymap) validate() error {
	contexts := map[Context]bool{Global: true}
	seen := map[string]bool{}
	for _, b := range k.bindings {
		contexts[b.Context] = true
		id := string(b.Context) + "\x00" + strings.Join(b.Keys, "\x00")
		if seen[id] {
			return fmt.Errorf("keymap: %s bound twice in context %s", k.Display(b.Keys), b.Context)
		}
		seen[id] = true
	}
	for ctx := range contexts {
		scope := k.scope(ctx)
		for _, a := range scope {
			for _, b := range scope {
				if len(a.Keys) < len(b.Keys) && hasPrefix(b.Keys, a.Keys) {
					return fmt.Errorf("keymap: %s (%s) is a prefix of %s (%s) in context %s",
						k.Display(a.Keys), a.Action, k.Display(b.Keys), b.Action, ctx)
				}
			}
		}
	}
	return nil
}

// scope returns the bindings visible in ctx: ctx first, then unshadowed globals.
func (k *Keymap) scope(ctx Context) []Binding {
	var out []Binding
	seen := map[string]bool{}
	for _, c := range []Context{ctx, Global} {
		for _, b := range k.bindings {
			if b.Context != c {
				continue
			}
			id := strings.Join(b.Keys, "\x00")
			if seen[id] {
				continue
			}
			seen[id] = true
			out = append(out, b)
		}
		if ctx == Global {
			break
		}
	}
	return out
}

// Resolve looks up seq in ctx.
func (k *Keymap) Resolve(ctx Context, seq []string) (Action, Result) {
	pending := false
	for _, b := range k.scope(ctx) {
		if slices.Equal(b.Keys, seq) {
			return b.Action, Matched
		}
		if hasPrefix(b.Keys, seq) {
			pending = true
		}
	}
	if pending {
		return "", Pending
	}
	return "", NoMatch
}

// Continuations lists what can follow prefix in ctx, sorted by key.
func (k *Keymap) Continuations(ctx Context, prefix []string) []Hint {
	var out []Hint
	for _, b := range k.scope(ctx) {
		if len(b.Keys) > len(prefix) && hasPrefix(b.Keys, prefix) {
			out = append(out, Hint{Keys: k.Display(b.Keys[len(prefix):]), Desc: b.Desc})
		}
	}
	sort.Slice(out, func(i, j int) bool { return out[i].Keys < out[j].Keys })
	return out
}

// Help lists bindings visible in ctx, merging aliases of the same action.
func (k *Keymap) Help(ctx Context) []HelpEntry {
	var out []HelpEntry
	index := map[string]int{}
	for _, b := range k.scope(ctx) {
		id := string(b.Context) + "\x00" + string(b.Action)
		if i, ok := index[id]; ok {
			out[i].Keys += ", " + k.Display(b.Keys)
			continue
		}
		index[id] = len(out)
		out = append(out, HelpEntry{Context: b.Context, Keys: k.Display(b.Keys), Desc: b.Desc})
	}
	return out
}

// Display renders keys vim-style: "gg", "<space>s", "<ctrl+d>".
func (k *Keymap) Display(keys []string) string {
	var b strings.Builder
	for _, key := range keys {
		if len([]rune(key)) == 1 {
			b.WriteString(key)
		} else {
			b.WriteString("<" + key + ">")
		}
	}
	return b.String()
}

func hasPrefix(keys, prefix []string) bool {
	return len(keys) >= len(prefix) && slices.Equal(keys[:len(prefix)], prefix)
}
```

```go
// Package uitest builds Bubble Tea key events for tests.
package uitest

import (
	"strings"
	"unicode/utf8"

	tea "charm.land/bubbletea/v2"
)

var special = map[string]tea.KeyPressMsg{
	"enter":     {Code: tea.KeyEnter},
	"esc":       {Code: tea.KeyEscape},
	"tab":       {Code: tea.KeyTab},
	"shift+tab": {Code: tea.KeyTab, Mod: tea.ModShift},
	"backspace": {Code: tea.KeyBackspace},
	"space":     {Code: tea.KeySpace, Text: " "},
	"up":        {Code: tea.KeyUp},
	"down":      {Code: tea.KeyDown},
	"left":      {Code: tea.KeyLeft},
	"right":     {Code: tea.KeyRight},
}

// Key builds a key press whose String() is s, e.g. "j", "G", "ctrl+d", "enter".
func Key(s string) tea.KeyPressMsg {
	if k, ok := special[s]; ok {
		return k
	}
	if rest, ok := strings.CutPrefix(s, "ctrl+"); ok {
		r, _ := utf8.DecodeRuneInString(rest)
		return tea.KeyPressMsg{Code: r, Mod: tea.ModCtrl}
	}
	r, _ := utf8.DecodeRuneInString(s)
	return tea.KeyPressMsg{Code: r, Text: s}
}

// Type returns one key press per rune of s.
func Type(s string) []tea.KeyPressMsg {
	var out []tea.KeyPressMsg
	for _, r := range s {
		out = append(out, Key(string(r)))
	}
	return out
}
```

```go
// Package overlay draws popups (help, prompts, cell detail) over the main UI.
package overlay

import (
	"strings"

	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
	"github.com/charmbracelet/x/ansi"
)

// Overlay is a modal popup. HandleKey returns a literal nil Overlay to close.
type Overlay interface {
	HandleKey(msg tea.KeyPressMsg) (Overlay, tea.Cmd)
	View(width, height int) string
}

const reset = "\x1b[0m"

// PlaceAt draws box over base with its top-left corner at (x, y).
func PlaceAt(base, box string, x, y int) string {
	lines := strings.Split(base, "\n")
	for i, bl := range strings.Split(box, "\n") {
		row := y + i
		if row < 0 || row >= len(lines) {
			continue
		}
		line := lines[row]
		left := ansi.Truncate(line, x, "")
		if pad := x - ansi.StringWidth(left); pad > 0 {
			left += strings.Repeat(" ", pad)
		}
		right := ansi.TruncateLeft(line, x+ansi.StringWidth(bl), "")
		lines[row] = left + reset + bl + reset + right
	}
	return strings.Join(lines, "\n")
}

// Center draws box in the middle of a width x height base.
func Center(base, box string, width, height int) string {
	x := max((width-lipgloss.Width(box))/2, 0)
	y := max((height-lipgloss.Height(box))/2, 0)
	return PlaceAt(base, box, x, y)
}

func emit(msg tea.Msg) tea.Cmd { return func() tea.Msg { return msg } }
```

```go
package overlay

import tea "charm.land/bubbletea/v2"

// field is a minimal single-line text input (append-only cursor).
type field struct{ r []rune }

func (f *field) handle(msg tea.KeyPressMsg) {
	switch msg.String() {
	case "backspace":
		if len(f.r) > 0 {
			f.r = f.r[:len(f.r)-1]
		}
	case "ctrl+u":
		f.r = nil
	default:
		if msg.Text != "" && !msg.Mod.Contains(tea.ModCtrl) {
			f.r = append(f.r, []rune(msg.Text)...)
		}
	}
}

func (f *field) String() string { return string(f.r) }
func (f *field) Len() int       { return len(f.r) }
```

```go
package overlay

import (
	"fmt"

	tea "charm.land/bubbletea/v2"

	"github.com/shepherd-yun/sqlvim/internal/ui/keymap"
	"github.com/shepherd-yun/sqlvim/internal/ui/textutil"
	"github.com/shepherd-yun/sqlvim/internal/ui/theme"
)

// Help lists the key bindings of the focused context.
type Help struct {
	lines  []string
	offset int
}

// NewHelp groups entries by context.
func NewHelp(entries []keymap.HelpEntry) *Help {
	var lines []string
	var last keymap.Context
	for _, e := range entries {
		if e.Context != last {
			if len(lines) > 0 {
				lines = append(lines, "")
			}
			lines = append(lines, theme.Header.Render(string(e.Context)))
			last = e.Context
		}
		lines = append(lines, fmt.Sprintf("  %-16s %s", e.Keys, e.Desc))
	}
	return &Help{lines: lines}
}

func (h *Help) HandleKey(msg tea.KeyPressMsg) (Overlay, tea.Cmd) {
	switch msg.String() {
	case "esc", "q", "?":
		return nil, nil
	case "j", "down":
		h.offset++
	case "k", "up":
		h.offset--
	}
	h.offset = textutil.Clamp(h.offset, 0, len(h.lines)-1)
	return h, nil
}

func (h *Help) View(width, height int) string {
	return scrollBox("Help · j/k scroll · esc close", h.lines, &h.offset, width, height, 70)
}

// scrollBox frames lines[offset:] in a box sized to fit within width x height.
func scrollBox(title string, lines []string, offset *int, width, height, maxWidth int) string {
	w := max(min(width-4, maxWidth), 10)
	h := max(min(height-2, len(lines)+2), 3)
	*offset = textutil.Clamp(*offset, 0, max(len(lines)-(h-2), 0))
	end := min(*offset+h-2, len(lines))
	body := ""
	for i, l := range lines[*offset:end] {
		if i > 0 {
			body += "\n"
		}
		body += l
	}
	return theme.Box(title, body, w, h, true)
}
```

```go
package overlay

import (
	"bytes"
	"encoding/json"
	"strings"

	tea "charm.land/bubbletea/v2"

	"github.com/shepherd-yun/sqlvim/internal/ui/textutil"
)

// Detail shows one cell value in full; JSON is pretty-printed.
type Detail struct {
	title  string
	text   string
	offset int
}

// NewDetail builds the popup for a cell.
func NewDetail(title, text string) *Detail {
	return &Detail{title: title, text: prettyJSON(text)}
}

func prettyJSON(s string) string {
	t := strings.TrimSpace(s)
	if !(strings.HasPrefix(t, "{") || strings.HasPrefix(t, "[")) || !json.Valid([]byte(t)) {
		return s
	}
	var b bytes.Buffer
	if err := json.Indent(&b, []byte(t), "", "  "); err != nil {
		return s
	}
	return b.String()
}

func (d *Detail) HandleKey(msg tea.KeyPressMsg) (Overlay, tea.Cmd) {
	switch msg.String() {
	case "esc", "q", "K":
		return nil, nil
	case "j", "down":
		d.offset++
	case "k", "up":
		d.offset--
	}
	return d, nil
}

func (d *Detail) View(width, height int) string {
	inner := max(min(width-4, 100), 10) - 2
	var lines []string
	for _, l := range strings.Split(d.text, "\n") {
		lines = append(lines, textutil.Wrap(textutil.Sanitize(l), inner)...)
	}
	return scrollBox(d.title+" · esc close", lines, &d.offset, width, height, 100)
}
```

```go
	input field
```

```go
// CmdLine is the vim-style ":" command line drawn in the status bar.

		return nil, emit(CommandMsg{Text: c.input.String()})

	return c, nil
```

```go
// Package tree is the connection / database / schema / table navigator.
package tree

import (
	"strings"

	"github.com/shepherd-yun/sqlvim/internal/db"
	"github.com/shepherd-yun/sqlvim/internal/ui/textutil"
	"github.com/shepherd-yun/sqlvim/internal/ui/theme"
)

// Kind is the type of a tree node.
type Kind int

const (
	KindConn Kind = iota
	KindDatabase
	KindSchema
	KindTable
	KindView
)

// Node is one tree entry. Children are loaded lazily by the caller.
type Node struct {
	Kind     Kind
	Name     string
	Conn     string
	NS       db.Namespace
	Prod     bool
	Expanded bool
	Loaded   bool
	Loading  bool
	Err      string
	Children []*Node

	parent *Node
}

func (n *Node) Parent() *Node        { return n.parent }
func (n *Node) Expandable() bool     { return n.Kind <= KindSchema }
func (n *Node) IsTable() bool        { return n.Kind == KindTable || n.Kind == KindView }
func (n *Node) TableRef() db.TableRef { return db.TableRef{Namespace: n.NS, Name: n.Name} }

type row struct {
	node  *Node
	depth int
}

// Model holds the tree and the cursor.
type Model struct {
	roots    []*Node
	selected *Node
	offset   int
	height   int
}

// New builds a tree from root nodes.
func New(roots []*Node) *Model {
	m := &Model{roots: roots}
	if len(roots) > 0 {
		m.selected = roots[0]
	}
	return m
}

// SetHeight sets the number of visible rows.
func (m *Model) SetHeight(h int) { m.height = max(h, 0) }

func (m *Model) visible() []row {
	var out []row
	var walk func(nodes []*Node, depth int)
	walk = func(nodes []*Node, depth int) {
		for _, n := range nodes {
			out = append(out, row{node: n, depth: depth})
			if n.Expanded {
				walk(n.Children, depth+1)
			}
		}
	}
	walk(m.roots, 0)
	return out
}

// cursor returns the index of the selected node, falling back to the nearest
// visible ancestor, and re-anchors the selection there.
func (m *Model) cursor(rows []row) int {
	for n := m.selected; n != nil; n = n.parent {
		for i, r := range rows {
			if r.node == n {
				m.selected = n
				return i
			}
		}
	}
	if len(rows) > 0 {
		m.selected = rows[0].node
	}
	return 0
}

// Selected returns the node under the cursor, or nil for an empty tree.
func (m *Model) Selected() *Node {
	rows := m.visible()
	if len(rows) == 0 {
		return nil
	}
	return rows[m.cursor(rows)].node
}

// Move moves the cursor by delta rows.
func (m *Model) Move(delta int) {
	rows := m.visible()
	if len(rows) == 0 {
		return
	}
	m.selected = rows[textutil.Clamp(m.cursor(rows)+delta, 0, len(rows)-1)].node
}

func (m *Model) Top()    { m.Move(-1 << 30) }
func (m *Model) Bottom() { m.Move(1 << 30) }

// HalfPage moves half the visible height in direction dir (+1 / -1).
func (m *Model) HalfPage(dir int) { m.Move(dir * max(m.height/2, 1)) }

// Expand opens the selected node. It returns true when the caller must load children.
func (m *Model) Expand() bool {
	n := m.Selected()
	if n == nil || !n.Expandable() || n.Loading {
		return false
	}
	if n.Loaded {
		n.Expanded = true
		return false
	}
	m.StartLoad(n)
	return true
}

// Collapse closes the selected node, or selects its parent if already closed.
func (m *Model) Collapse() {
	n := m.Selected()
	if n == nil {
		return
	}
	if n.Expanded {
		n.Expanded = false
		return
	}
	if n.parent != nil {
		m.selected = n.parent
	}
}

// StartLoad marks n as loading (used for first load, retry and refresh).
func (m *Model) StartLoad(n *Node) {
	n.Loading, n.Expanded, n.Err = true, true, ""
}

// CancelLoad undoes StartLoad without recording an error.
func (m *Model) CancelLoad(n *Node) {
	n.Loading = false
	n.Expanded = n.Loaded
}

// SetChildren records the result of a load.
func (m *Model) SetChildren(n *Node, children []*Node, err error) {
	n.Loading = false
	if err != nil {
		n.Err, n.Expanded, n.Loaded = err.Error(), false, false
		return
	}
	for _, c := range children {
		c.parent = n
	}
	n.Children, n.Loaded, n.Expanded, n.Err = children, true, true, ""
}

// ResetConn forgets everything loaded under the named connection.
func (m *Model) ResetConn(conn string) {
	for _, r := range m.roots {
		if r.Conn == conn {
			r.Children, r.Loaded, r.Expanded, r.Loading, r.Err = nil, false, false, false, ""
		}
	}
}

// View renders the visible window of the tree.
func (m *Model) View(width int, focused bool) string {
	rows := m.visible()
	if len(rows) == 0 || m.height == 0 || width <= 0 {
		return ""
	}
	cur := m.cursor(rows)
	if cur < m.offset {
		m.offset = cur
	}
	if cur >= m.offset+m.height {
		m.offset = cur - m.height + 1
	}
	m.offset = textutil.Clamp(m.offset, 0, max(len(rows)-m.height, 0))

	end := min(m.offset+m.height, len(rows))
	lines := make([]string, 0, end-m.offset)
	for i := m.offset; i < end; i++ {
		n := rows[i].node
		label := strings.Repeat("  ", rows[i].depth) + icon(n) + " " + textutil.Sanitize(n.Name)
		if i == cur {
			if focused {
				label = theme.Selected.Render(label)
			} else {
				label = theme.SelectedDim.Render(label)
			}
		}
		if n.Prod {
			label += " " + theme.Prod.Render("[PROD]")
		}
		if n.Loading {
			label += " " + theme.Muted.Render("loading…")
		}
		if n.Err != "" {
			label += " " + theme.Error.Render("✗ "+textutil.Sanitize(n.Err))
		}
		lines = append(lines, textutil.Fit(label, width))
	}
	return strings.Join(lines, "\n")
}

func icon(n *Node) string {
	switch {
	case n.Kind == KindTable:
		return "•"
	case n.Kind == KindView:
		return "◦"
	case n.Expanded:
		return "▾"
	default:
		return "▸"
	}
}
```

```go
// Package grid renders a result set as a navigable table.
package grid

import (
	"strings"

	"charm.land/lipgloss/v2"
	"github.com/charmbracelet/x/ansi"

	"github.com/shepherd-yun/sqlvim/internal/db"
	"github.com/shepherd-yun/sqlvim/internal/ui/textutil"
	"github.com/shepherd-yun/sqlvim/internal/ui/theme"
)

const (
	sep      = " │ "
	sepWidth = 3
	// widthSample bounds the bytes measured per cell so huge values stay cheap.
	widthSample = 1024
)

// Model is a table with a cell cursor.
type Model struct {
	rs             *db.ResultSet
	maxColumnWidth int
	widths         []int
	row, col       int
	rowOff, colOff int
	width, height  int
}

// New builds a grid; rs may be nil.
func New(rs *db.ResultSet, maxColumnWidth int) *Model {
	if maxColumnWidth <= 0 {
		maxColumnWidth = 40
	}
	m := &Model{maxColumnWidth: maxColumnWidth}
	m.SetResult(rs)
	return m
}

// SetResult replaces the data and resets the cursor.
func (m *Model) SetResult(rs *db.ResultSet) {
	m.rs = rs
	m.row, m.col, m.rowOff, m.colOff = 0, 0, 0, 0
	m.widths = nil
	if rs == nil {
		return
	}
	m.widths = make([]int, len(rs.Columns))
	for c, col := range rs.Columns {
		w := ansi.StringWidth(textutil.Sanitize(col.Name))
		for _, r := range rs.Rows {
			w = max(w, ansi.StringWidth(cellText(r[c])))
		}
		m.widths[c] = textutil.Clamp(w, 1, m.maxColumnWidth)
	}
}

// SetSize sets the drawing area.
func (m *Model) SetSize(w, h int) { m.width, m.height = max(w, 0), max(h, 0) }

// RowCount returns the number of data rows.
func (m *Model) RowCount() int {
	if m.rs == nil {
		return 0
	}
	return len(m.rs.Rows)
}

func (m *Model) colCount() int { return len(m.widths) }

// Move moves the cursor, clamped to the data.
func (m *Model) Move(dRow, dCol int) {
	m.row = textutil.Clamp(m.row+dRow, 0, m.RowCount()-1)
	m.col = textutil.Clamp(m.col+dCol, 0, m.colCount()-1)
}

func (m *Model) Top()       { m.row = 0 }
func (m *Model) Bottom()    { m.row = max(m.RowCount()-1, 0) }
func (m *Model) LineStart() { m.col = 0 }
func (m *Model) LineEnd()   { m.col = max(m.colCount()-1, 0) }

// HalfPage moves half the visible rows in direction dir.
func (m *Model) HalfPage(dir int) { m.Move(dir*max(m.bodyHeight()/2, 1), 0) }

// Cell returns the cell under the cursor.
func (m *Model) Cell() (db.ColumnInfo, db.Value, bool) {
	if m.RowCount() == 0 || m.colCount() == 0 {
		return db.ColumnInfo{}, db.Value{}, false
	}
	return m.rs.Columns[m.col], m.rs.Rows[m.row][m.col], true
}

func (m *Model) bodyHeight() int { return max(m.height-2, 0) }

func (m *Model) spanWidth(from, to int) int {
	w := 0
	for c := from; c <= to; c++ {
		w += m.widths[c]
		if c > from {
			w += sepWidth
		}
	}
	return w
}

func (m *Model) scrollIntoView() {
	if body := m.bodyHeight(); body > 0 {
		if m.row < m.rowOff {
			m.rowOff = m.row
		}
		if m.row >= m.rowOff+body {
			m.rowOff = m.row - body + 1
		}
	}
	if m.col < m.colOff {
		m.colOff = m.col
	}
	for m.colOff < m.col && m.spanWidth(m.colOff, m.col) > m.width {
		m.colOff++
	}
}

func (m *Model) visibleColumns() []int {
	var cols []int
	used := 0
	for c := m.colOff; c < m.colCount(); c++ {
		w := m.widths[c]
		if len(cols) > 0 {
			w += sepWidth
			if used+w > m.width {
				break
			}
		}
		cols = append(cols, c)
		used += w
	}
	return cols
}

// View renders exactly the visible part of the grid, each line m.width wide.
func (m *Model) View(focused bool) string {
	if m.width <= 0 || m.height <= 0 {
		return ""
	}
	if m.rs == nil || m.colCount() == 0 {
		return textutil.Fit(theme.Muted.Render("(no columns)"), m.width)
	}
	m.scrollIntoView()
	cols := m.visibleColumns()
	sepStyled := theme.Muted.Render(sep)

	header := make([]string, len(cols))
	rule := make([]string, len(cols))
	for i, c := range cols {
		header[i] = theme.Header.Render(textutil.Fit(textutil.Sanitize(m.rs.Columns[c].Name), m.widths[c]))
		rule[i] = strings.Repeat("─", m.widths[c])
	}
	lines := []string{
		strings.Join(header, sepStyled),
		theme.Muted.Render(strings.Join(rule, "─┼─")),
	}
	if m.RowCount() == 0 {
		lines = append(lines, theme.Muted.Render("(0 rows)"))
	}
	for r := m.rowOff; r < m.RowCount() && len(lines) < m.height; r++ {
		cells := make([]string, len(cols))
		for i, c := range cols {
			v := m.rs.Rows[r][c]
			style := lipgloss.NewStyle()
			if v.Null {
				style = theme.Null
			}
			if r == m.row && c == m.col {
				if focused {
					style = style.Reverse(true)
				} else {
					style = style.Underline(true)
				}
			}
			cells[i] = style.Render(textutil.Fit(cellText(v), m.widths[c]))
		}
		lines = append(lines, strings.Join(cells, sepStyled))
	}
	if len(lines) > m.height {
		lines = lines[:m.height]
	}
	for i, l := range lines {
		lines[i] = textutil.Fit(l, m.width)
	}
	return strings.Join(lines, "\n")
}

func cellText(v db.Value) string {
	if v.Null {
		return "NULL"
	}
	s := v.Text
	if len(s) > widthSample {
		s = s[:widthSample]
	}
	return textutil.Sanitize(s)
}
```

```go
// Package structure shows a table's columns, indexes, foreign keys and DDL.
package structure

import (
	"strings"

	"github.com/shepherd-yun/sqlvim/internal/db"
	"github.com/shepherd-yun/sqlvim/internal/ui/textutil"
	"github.com/shepherd-yun/sqlvim/internal/ui/theme"
)

// Data is everything the view shows.
type Data struct {
	Columns     []db.Column
	Indexes     []db.Index
	ForeignKeys []db.ForeignKey
	DDL         string
}

// Section is one sub-page.
type Section int

const (
	Columns Section = iota
	Indexes
	ForeignKeys
	DDL
	sectionCount
)

var sectionNames = [...]string{"Columns", "Indexes", "Foreign Keys", "DDL"}

// Model is the structure view state.
type Model struct {
	data          Data
	section       Section
	offset        int
	width, height int
}

func New() *Model { return &Model{} }

func (m *Model) SetData(d Data)       { m.data, m.offset = d, 0 }
func (m *Model) SetSize(w, h int)     { m.width, m.height = max(w, 0), max(h, 0) }
func (m *Model) Section() Section     { return m.section }
func (m *Model) NextSection()         { m.section, m.offset = (m.section+1)%sectionCount, 0 }
func (m *Model) PrevSection()         { m.section, m.offset = (m.section+sectionCount-1)%sectionCount, 0 }
func (m *Model) Top()                 { m.offset = 0 }
func (m *Model) Bottom()              { m.offset = m.maxOffset() }
func (m *Model) HalfPage(dir int)     { m.Scroll(dir * max(m.bodyHeight()/2, 1)) }
func (m *Model) Scroll(delta int)     { m.offset = textutil.Clamp(m.offset+delta, 0, m.maxOffset()) }
func (m *Model) bodyHeight() int      { return max(m.height-2, 0) }
func (m *Model) maxOffset() int       { return max(len(m.lines())-m.bodyHeight(), 0) }

func (m *Model) lines() []string {
	switch m.section {
	case Columns:
		rows := make([][]string, len(m.data.Columns))
		for i, c := range m.data.Columns {
			key, null, def := "", "NOT NULL", ""
			if c.PrimaryKey {
				key = "PK"
			}
			if c.Nullable {
				null = "NULL"
			}
			if c.Default != nil {
				def = *c.Default
			}
			rows[i] = []string{key, c.Name, c.Type, null, def, c.Comment}
		}
		return textutil.Table([]string{"", "Name", "Type", "Null", "Default", "Comment"}, rows)
	case Indexes:
		rows := make([][]string, len(m.data.Indexes))
		for i, idx := range m.data.Indexes {
			kind := "INDEX"
			switch {
			case idx.Primary:
				kind = "PRIMARY"
			case idx.Unique:
				kind = "UNIQUE"
			}
			rows[i] = []string{idx.Name, kind, strings.Join(idx.Columns, ", ")}
		}
		return textutil.Table([]string{"Name", "Kind", "Columns"}, rows)
	case ForeignKeys:
		rows := make([][]string, len(m.data.ForeignKeys))
		for i, fk := range m.data.ForeignKeys {
			ref := fk.RefTable.Name
			if ns := fk.RefTable.Schema; ns != "" {
				ref = ns + "." + ref
			} else if ns := fk.RefTable.Database; ns != "" {
				ref = ns + "." + ref
			}
			rows[i] = []string{fk.Name, strings.Join(fk.Columns, ", "), ref + "(" + strings.Join(fk.RefColumns, ", ") + ")"}
		}
		return textutil.Table([]string{"Name", "Columns", "References"}, rows)
	default:
		return strings.Split(m.data.DDL, "\n")
	}
}

// View renders the section header and the visible body lines.
func (m *Model) View() string {
	if m.width <= 0 || m.height <= 0 {
		return ""
	}
	tabs := make([]string, len(sectionNames))
	for i, name := range sectionNames {
		if Section(i) == m.section {
			tabs[i] = theme.ActiveTab.Render("[" + name + "]")
		} else {
			tabs[i] = theme.Muted.Render(" " + name + " ")
		}
	}
	out := []string{textutil.Fit(strings.Join(tabs, " ")+theme.Muted.Render("  [ ] switch"), m.width), ""}

	lines := m.lines()
	m.offset = textutil.Clamp(m.offset, 0, m.maxOffset())
	end := min(m.offset+m.bodyHeight(), len(lines))
	for _, l := range lines[m.offset:end] {
		out = append(out, textutil.Fit(textutil.Sanitize(l), m.width))
	}
	if len(out) > m.height {
		out = out[:m.height]
	}
	return strings.Join(out, "\n")
}
```

```go
package ui

import (
	"context"
	"errors"
	"fmt"
	"strings"
	"time"

	tea "charm.land/bubbletea/v2"

	"github.com/shepherd-yun/sqlvim/internal/config"
	"github.com/shepherd-yun/sqlvim/internal/db"
	"github.com/shepherd-yun/sqlvim/internal/ui/keymap"
	"github.com/shepherd-yun/sqlvim/internal/ui/overlay"
	"github.com/shepherd-yun/sqlvim/internal/ui/panes/grid"
	"github.com/shepherd-yun/sqlvim/internal/ui/panes/structure"
	"github.com/shepherd-yun/sqlvim/internal/ui/textutil"
	"github.com/shepherd-yun/sqlvim/internal/ui/theme"
)

const (
	loadTimeout  = 15 * time.Second
	queryTimeout = 30 * time.Second
)

// tab is one entry in the right-hand tab area.
type tab interface {
	ID() int
	Title() string
	Key() string
	Context() keymap.Context
	Target() (conn string, ref db.TableRef)
	Init() tea.Cmd
	SetSize(w, h int)
	Handle(a keymap.Action) tea.Cmd
	Update(msg tea.Msg) tea.Cmd
	View(focused bool) string
}

// tabMsg is an async result addressed to one tab.
type tabMsg interface{ TabID() int }

// openOverlayMsg asks the root model to show a popup.
type openOverlayMsg struct{ overlay overlay.Overlay }

func openOverlay(o overlay.Overlay) tea.Cmd {
	return func() tea.Msg { return openOverlayMsg{overlay: o} }
}

func tableKey(conn string, ref db.TableRef) string {
	return strings.Join([]string{conn, ref.Database, ref.Schema, ref.Name}, "\x00")
}

func dataTabKey(conn string, ref db.TableRef) string      { return "data\x00" + tableKey(conn, ref) }
func structureTabKey(conn string, ref db.TableRef) string { return "struct\x00" + tableKey(conn, ref) }

// ---- data tab ----

type pageLoadedMsg struct {
	tabID   int
	page    int
	rs      *db.ResultSet
	hasMore bool
	orderBy []string
	err     error
}

func (m pageLoadedMsg) TabID() int { return m.tabID }

type dataTab struct {
	id       int
	backend  Backend
	conn     string
	ref      db.TableRef
	pageSize int
	grid     *grid.Model

	page       int
	hasMore    bool
	loading    bool
	err        string
	orderBy    []string
	orderKnown bool
	height     int
}

func newDataTab(id int, b Backend, conn string, ref db.TableRef, s config.Settings) *dataTab {
	return &dataTab{
		id: id, backend: b, conn: conn, ref: ref, pageSize: s.PageSize,
		grid: grid.New(nil, s.MaxColumnWidth),
	}
}

func (t *dataTab) ID() int                         { return t.id }
func (t *dataTab) Title() string                   { return t.ref.Name }
func (t *dataTab) Key() string                     { return dataTabKey(t.conn, t.ref) }
func (t *dataTab) Context() keymap.Context         { return keymap.Grid }
func (t *dataTab) Target() (string, db.TableRef)   { return t.conn, t.ref }
func (t *dataTab) Init() tea.Cmd                   { return t.load(0) }

func (t *dataTab) SetSize(w, h int) {
	t.height = max(h, 0)
	t.grid.SetSize(w, max(h-1, 0))
}

func (t *dataTab) load(page int) tea.Cmd {
	t.loading = true
	id, b, conn, ref, size := t.id, t.backend, t.conn, t.ref, t.pageSize
	orderBy, known := t.orderBy, t.orderKnown
	return func() tea.Msg {
		ctx, cancel := context.WithTimeout(context.Background(), queryTimeout)
		defer cancel()
		msg := pageLoadedMsg{tabID: id, page: page, orderBy: orderBy}
		c, err := b.Conn(ctx, conn, ref.Database)
		if err != nil {
			msg.err = err
			return msg
		}
		if !known {
			cols, err := c.Meta().Columns(ctx, ref)
			if err != nil {
				msg.err = err
				return msg
			}
			msg.orderBy = primaryKey(cols)
		}
		msg.rs, msg.hasMore, msg.err = fetchPage(ctx, c, ref, msg.orderBy, page, size)
		return msg
	}
}

func primaryKey(cols []db.Column) []string {
	var out []string
	for _, c := range cols {
		if c.PrimaryKey {
			out = append(out, c.Name)
		}
	}
	return out
}

func fetchPage(ctx context.Context, c db.Conn, ref db.TableRef, orderBy []string, page, size int) (*db.ResultSet, bool, error) {
	d := c.Dialect()
	q := "SELECT * FROM " + d.QualifiedName(ref)
	if len(orderBy) > 0 {
		quoted := make([]string, len(orderBy))
		for i, col := range orderBy {
			quoted[i] = d.QuoteIdent(col)
		}
		q += " ORDER BY " + strings.Join(quoted, ", ")
	}
	rs, err := c.Query(ctx, d.Paginate(q, size+1, page*size))
	if err != nil {
		return nil, false, err
	}
	hasMore := len(rs.Rows) > size
	if hasMore {
		rs.Rows = rs.Rows[:size]
	}
	return rs, hasMore, nil
}

func (t *dataTab) Update(msg tea.Msg) tea.Cmd {
	m, ok := msg.(pageLoadedMsg)
	if !ok {
		return nil
	}
	t.loading = false
	if m.err != nil {
		t.err = m.err.Error()
		return nil
	}
	t.err, t.page, t.hasMore = "", m.page, m.hasMore
	t.orderBy, t.orderKnown = m.orderBy, true
	t.grid.SetResult(m.rs)
	return nil
}

func (t *dataTab) Handle(a keymap.Action) tea.Cmd {
	g := t.grid
	switch a {
	case keymap.Down:
		g.Move(1, 0)
	case keymap.Up:
		g.Move(-1, 0)
	case keymap.Left:
		g.Move(0, -1)
	case keymap.Right:
		g.Move(0, 1)
	case keymap.Top:
		g.Top()
	case keymap.Bottom:
		g.Bottom()
	case keymap.HalfPageDown:
		g.HalfPage(1)
	case keymap.HalfPageUp:
		g.HalfPage(-1)
	case keymap.LineStart:
		g.LineStart()
	case keymap.LineEnd:
		g.LineEnd()
	case keymap.NextPage:
		if t.hasMore && !t.loading {
			return t.load(t.page + 1)
		}
	case keymap.PrevPage:
		if t.page > 0 && !t.loading {
			return t.load(t.page - 1)
		}
	case keymap.Refresh:
		if !t.loading {
			t.orderKnown = false
			return t.load(t.page)
		}
	case keymap.CellDetail:

			text := v.Text
			if v.Null {
				text = "NULL"
			}
			return openOverlay(overlay.NewDetail(col.Name, text))
		}
	}
	return nil
}

	case t.loading:

	}

			return structureLoadedMsg{tabID: id, err: err}

func fetchStructure(ctx context.Context, m db.MetaReader, ref db.TableRef) (structure.Data, error) {
	var d structure.Data
	var err error
	if d.DDL, err = m.DDL(ctx, ref); err != nil {
		if errors.Is(err, db.ErrNotFound) {
			return d, fmt.Errorf("table %s not found", ref.Name)
		}
		return d, err
	}
	if d.Columns, err = m.Columns(ctx, ref); err != nil {
		return d, err
	}
	if d.Indexes, err = m.Indexes(ctx, ref); err != nil {
		return d, err
	}
	d.ForeignKeys, err = m.ForeignKeys(ctx, ref)
	return d, err
}

	return nil
```
```go
}

// drain runs cmd and feeds every resulting message back into m.
// It reports whether a tea.QuitMsg was produced.
func drain(t *testing.T, m *Model, cmd tea.Cmd) (quit bool) {
	t.Helper()
	queue := []tea.Cmd{cmd}
	for steps := 0; len(queue) > 0; steps++ {
		if steps > 100 {
			t.Fatal("drain: too many steps")
		}
		c := queue[0]
		queue = queue[1:]
		if c == nil {
			continue
		}
		switch msg := c().(type) {
		case nil:
		case tea.BatchMsg:
			queue = append(queue, msg...)
		case tea.QuitMsg:
			quit = true
		default:
			_, next := m.Update(msg)
			queue = append(queue, next)
		}
	}
	return quit
}
```

```go
package ui

import (
	"context"
	"fmt"
	"log/slog"

	tea "charm.land/bubbletea/v2"

	"github.com/shepherd-yun/sqlvim/internal/db"
	"github.com/shepherd-yun/sqlvim/internal/ui/panes/tree"
)

type childrenLoadedMsg struct {
	node     *tree.Node
	children []*tree.Node
	err      error
}

// loadChildren copies what it needs from n so the goroutine never reads
// fields the UI goroutine may be writing.
func loadChildren(b Backend, n *tree.Node) tea.Cmd {
	kind, connName, ns := n.Kind, n.Conn, n.NS
	return func() tea.Msg {
		ctx, cancel := context.WithTimeout(context.Background(), loadTimeout)
		defer cancel()
		children, err := fetchChildren(ctx, b, kind, connName, ns)
		if err != nil {
			slog.Warn("load tree children", "conn", connName, "database", ns.Database, "schema", ns.Schema, "err", err)
		}
		return childrenLoadedMsg{node: n, children: children, err: err}
	}
}

func fetchChildren(ctx context.Context, b Backend, kind tree.Kind, connName string, ns db.Namespace) ([]*tree.Node, error) {
	conn, err := b.Conn(ctx, connName, ns.Database)
	if err != nil {
		return nil, err
	}
	meta := conn.Meta()
	switch kind {
	case tree.KindConn:
		names, err := meta.Databases(ctx)
		if err != nil {
			return nil, err
		}
		nodes := make([]*tree.Node, len(names))
		for i, name := range names {
			nodes[i] = &tree.Node{Kind: tree.KindDatabase, Name: name, Conn: connName, NS: db.Namespace{Database: name}}
		}
		return nodes, nil
	case tree.KindDatabase:
		schemas, err := meta.Schemas(ctx, ns.Database)
		if err != nil {
			return nil, err
		}
		if len(schemas) == 0 {
			return tableNodes(ctx, meta, connName, ns)
		}
		nodes := make([]*tree.Node, len(schemas))
		for i, s := range schemas {
			nodes[i] = &tree.Node{Kind: tree.KindSchema, Name: s, Conn: connName, NS: db.Namespace{Database: ns.Database, Schema: s}}
		}
		return nodes, nil
	case tree.KindSchema:
		return tableNodes(ctx, meta, connName, ns)
	}
	return nil, fmt.Errorf("tree node kind %d has no children", kind)
}

func tableNodes(ctx context.Context, meta db.MetaReader, connName string, ns db.Namespace) ([]*tree.Node, error) {
	tables, err := meta.Tables(ctx, ns)
	if err != nil {
		return nil, err
	}
	nodes := make([]*tree.Node, len(tables))
	for i, t := range tables {
		kind := tree.KindTable
		if t.Kind == db.TableKindView {
			kind = tree.KindView
		}
		nodes[i] = &tree.Node{Kind: kind, Name: t.Name, Conn: connName, NS: ns}
	}
	return nodes, nil
}
```

```go
const defaultWhichKeyDelay = 300 * time.Millisecond

const welcome = `Select a table in the tree and press l or enter.

<space>s   table structure
?          help
:qa        quit`

type focus int

const (
	focusTree focus = iota
	focusTabs
)

type whichKeyMsg struct{ gen int }

// Model is the root Bubble Tea model.
type Model struct {
	backend  Backend
	settings config.Settings
	keys     *keymap.Keymap

	width, height int
	focus         focus
	showTree      bool
	tree          *tree.Model

	tabs      []tab
	active    int
	nextTabID int

	pending       []string
	pendingGen    int
	showWhichKey  bool
	whichKeyDelay time.Duration

	overlay   overlay.Overlay
	cmdline   *overlay.CmdLine
	retryNode *tree.Node

	status    string
	statusErr bool
}

// New builds the root model with one tree root per configured connection.
func New(b Backend, settings config.Settings) *Model {
	conns := b.Connections()
	roots := make([]*tree.Node, len(conns))
	for i, c := range conns {
		roots[i] = &tree.Node{Kind: tree.KindConn, Name: c.Name, Conn: c.Name, Prod: c.Env == config.EnvProd}
	}
	return &Model{
		backend:       b,
		settings:      settings,
		keys:          keymap.Default(),
		showTree:      true,
		tree:          tree.New(roots),
		whichKeyDelay: defaultWhichKeyDelay,
	}
}

func (m *Model) Init() tea.Cmd { return nil }

func (m *Model) View() tea.View {
	v := tea.NewView(m.render())
	v.AltScreen = true
	return v
}

func (m *Model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.width, m.height = msg.Width, msg.Height
		m.layout()
	case tea.KeyPressMsg:
		return m, m.handleKey(msg)
	case whichKeyMsg:
		if msg.gen == m.pendingGen && len(m.pending) > 0 {
			m.showWhichKey = true
		}
	case childrenLoadedMsg:
		return m, m.childrenLoaded(msg)
	case overlay.PasswordSubmittedMsg:
		return m, m.passwordSubmitted(msg)
	case overlay.CommandMsg:
		return m, m.runCommand(msg.Text)
	case openOverlayMsg:
		m.overlay = msg.overlay
	case tabMsg:
		if t := m.tabByID(msg.TabID()); t != nil {
			return m, t.Update(msg)
		}
	}
	return m, nil
}

// ---- keys ----

func (m *Model) handleKey(msg tea.KeyPressMsg) tea.Cmd {
	m.status, m.statusErr = "", false
	if m.cmdline != nil {
		next, cmd := m.cmdline.HandleKey(msg)
		if next == nil {
			m.cmdline = nil
		}
		return cmd
	}
	if m.overlay != nil {
		next, cmd := m.overlay.HandleKey(msg)
		m.overlay = next
		return cmd
	}
	key := msg.String()
	if key == "esc" && len(m.pending) > 0 {
		m.clearPending()
		return nil
	}
	m.pending = append(m.pending, key)
	action, res := m.keys.Resolve(m.context(), m.pending)
	switch res {
	case keymap.Pending:
		m.pendingGen++
		gen := m.pendingGen
		return tea.Tick(m.whichKeyDelay, func(time.Time) tea.Msg { return whichKeyMsg{gen: gen} })
	case keymap.Matched:
		m.clearPending()
		return m.dispatch(action)
	default:
		m.clearPending()
		return nil
	}
}

func (m *Model) clearPending() {
	m.pending = nil
	m.showWhichKey = false
}

func (m *Model) context() keymap.Context {
	if m.focus == focusTabs {
		if t := m.activeTab(); t != nil {
			return t.Context()
		}
	}
	return keymap.Tree
}

func (m *Model) dispatch(a keymap.Action) tea.Cmd {
	if n, ok := keymap.TabNumber(a); ok {
		m.gotoTab(n - 1)
		return nil
	}
	switch a {
	case keymap.Quit:
		return tea.Quit
	case keymap.CommandLine:
		m.cmdline = overlay.NewCmdLine()
	case keymap.Help:
		m.overlay = overlay.NewHelp(m.keys.Help(m.context()))
	case keymap.FocusNext, keymap.FocusPrev:
		if m.focus == focusTree {
			m.setFocus(focusTabs)
		} else {
			m.setFocus(focusTree)
		}
	case keymap.FocusLeft:
		m.setFocus(focusTree)
	case keymap.FocusRight:
		m.setFocus(focusTabs)
	case keymap.ToggleTree:
		m.toggleTree()
	case keymap.OpenStructure:
		return m.openStructureFromContext()
	default:
		if m.focus == focusTree {
			return m.treeAction(a)
		}
		if t := m.activeTab(); t != nil {
			return t.Handle(a)
		}
	}
	return nil
}

func (m *Model) treeAction(a keymap.Action) tea.Cmd {
	t := m.tree
	switch a {
	case keymap.Down:
		t.Move(1)
	case keymap.Up:
		t.Move(-1)
	case keymap.Top:
		t.Top()
	case keymap.Bottom:
		t.Bottom()
	case keymap.HalfPageDown:
		t.HalfPage(1)
	case keymap.HalfPageUp:
		t.HalfPage(-1)
	case keymap.Collapse:
		t.Collapse()
	case keymap.Expand:
		n := t.Selected()
		if n == nil {
			return nil
		}
		if n.IsTable() {
			return m.openData(n.Conn, n.TableRef())
		}
		if t.Expand() {
			return loadChildren(m.backend, n)
		}
	case keymap.Refresh:
		if n := t.Selected(); n != nil && n.Expandable() && !n.Loading {
			t.StartLoad(n)
			return loadChildren(m.backend, n)
		}
	case keymap.Password:
		if n := t.Selected(); n != nil {
			m.retryNode = nil
			m.overlay = overlay.NewPassword(n.Conn)
		}
	}
	return nil
}

func (m *Model) childrenLoaded(msg childrenLoadedMsg) tea.Cmd {
	if errors.Is(msg.err, session.ErrPasswordRequired) {
		m.tree.CancelLoad(msg.node)
		m.retryNode = msg.node
		m.overlay = overlay.NewPassword(msg.node.Conn)
		return nil
	}
	m.tree.SetChildren(msg.node, msg.children, msg.err)
	if msg.err != nil {
		m.setStatus(msg.err.Error(), true)
	}
	return nil
}

func (m *Model) passwordSubmitted(msg overlay.PasswordSubmittedMsg) tea.Cmd {
	if err := m.backend.SetPassword(msg.Conn, msg.Password, msg.Save); err != nil {
		m.setStatus("password kept for this session only: "+err.Error(), true)
	}
	node := m.retryNode
	m.retryNode = nil
	if node == nil || node.Conn != msg.Conn {
		m.tree.ResetConn(msg.Conn)
		return nil
	}
	m.tree.StartLoad(node)
	return loadChildren(m.backend, node)
}

func (m *Model) runCommand(text string) tea.Cmd {
	switch cmd := strings.TrimSpace(text); cmd {
	case "":
	case "q", "q!", "close":
		if len(m.tabs) == 0 {
			return tea.Quit
		}
		m.closeTab(m.active)
	case "qa", "qa!", "quit":
		return tea.Quit
	default:
		m.setStatus("unknown command: "+cmd, true)
	}
	return nil
}

func (m *Model) setStatus(s string, isErr bool) { m.status, m.statusErr = s, isErr }

// ---- focus and tabs ----

func (m *Model) setFocus(f focus) {
	switch f {
	case focusTree:
		if !m.showTree {
			m.showTree = true
			m.layout()
		}
		m.focus = focusTree
	case focusTabs:
		if len(m.tabs) > 0 {
			m.focus = focusTabs
		}
	}
}

func (m *Model) toggleTree() {
	if !m.showTree {
		m.showTree = true
	} else if len(m.tabs) > 0 {
		m.showTree = false
		m.focus = focusTabs
	}
	m.layout()
}

func (m *Model) activeTab() tab {
	if m.active < 0 || m.active >= len(m.tabs) {
		return nil
	}
	return m.tabs[m.active]
}

func (m *Model) tabByID(id int) tab {
	for _, t := range m.tabs {
		if t.ID() == id {
			return t
		}
	}
	return nil
}

func (m *Model) gotoTab(i int) {
	if i >= 0 && i < len(m.tabs) {
		m.active = i
		m.focus = focusTabs
	}
}

func (m *Model) closeTab(i int) {
	m.tabs = append(m.tabs[:i], m.tabs[i+1:]...)
	m.active = textutil.Clamp(m.active, 0, len(m.tabs)-1)
	if len(m.tabs) == 0 {
		m.showTree = true
		m.focus = focusTree
	}
	m.layout()
}

func (m *Model) openTab(key string, create func(id int) tab) tea.Cmd {
	for i, t := range m.tabs {
		if t.Key() == key {
			m.gotoTab(i)
			return nil
		}
	}
	m.nextTabID++
	t := create(m.nextTabID)
	m.tabs = append(m.tabs, t)
	m.active = len(m.tabs) - 1
	m.focus = focusTabs
	m.layout()
	return t.Init()
}

func (m *Model) openData(conn string, ref db.TableRef) tea.Cmd {
	return m.openTab(dataTabKey(conn, ref), func(id int) tab {
		return newDataTab(id, m.backend, conn, ref, m.settings)
	})
}

func (m *Model) openStructure(conn string, ref db.TableRef) tea.Cmd {
	return m.openTab(structureTabKey(conn, ref), func(id int) tab {
		return newStructureTab(id, m.backend, conn, ref)
	})
}

func (m *Model) openStructureFromContext() tea.Cmd {
	if m.focus == focusTree {
		if n := m.tree.Selected(); n != nil && n.IsTable() {
			return m.openStructure(n.Conn, n.TableRef())
		}
	} else if t := m.activeTab(); t != nil {
		conn, ref := t.Target()
		return m.openStructure(conn, ref)
	}
	m.setStatus("select a table first", true)
	return nil
}

// ---- layout and rendering ----

func (m *Model) treeWidth() int {
	if !m.showTree {
		return 0
	}
	return min(textutil.Clamp(m.width/4, 24, 40), m.width)
}

func (m *Model) bodyHeight() int { return max(m.height-1, 0) }

func (m *Model) layout() {
	inner := max(m.bodyHeight()-2, 0)
	m.tree.SetHeight(inner)
	tabsWidth := max(m.width-m.treeWidth()-2, 0)
	for _, t := range m.tabs {
		t.SetSize(tabsWidth, inner)
	}
}

func (m *Model) render() string {
	if m.width <= 0 || m.height <= 0 {
		return ""
	}
	bodyH := m.bodyHeight()
	treeW := m.treeWidth()
	var panes []string
	if treeW > 0 {
		focused := m.focus == focusTree
		panes = append(panes, theme.Box("Connections", m.tree.View(max(treeW-2, 0), focused), treeW, bodyH, focused))
	}
	if w := m.width - treeW; w > 0 {
		panes = append(panes, m.renderTabs(w, bodyH))
	}
	screen := lipgloss.JoinHorizontal(lipgloss.Top, panes...) + "\n" + m.renderStatus()
	if m.overlay != nil {
		screen = overlay.Center(screen, m.overlay.View(m.width, m.height), m.width, m.height)
	}
	if m.showWhichKey {
		box := overlay.WhichKey(m.keys.Continuations(m.context(), m.pending))
		x := max(m.width-lipgloss.Width(box), 0)
		y := max(m.height-1-lipgloss.Height(box), 0)
		screen = overlay.PlaceAt(screen, box, x, y)
	}
	return screen
}

func (m *Model) renderTabs(width, height int) string {
	t := m.activeTab()
	if t == nil {
		return theme.Box("sqlvim", welcome, width, height, false)
	}
	focused := m.focus == focusTabs
	return theme.Box(m.tabBar(), t.View(focused), width, height, focused)
}

func (m *Model) tabBar() string {
	labels := make([]string, len(m.tabs))
	for i, t := range m.tabs {
		label := fmt.Sprintf("[%d]%s", i+1, textutil.Sanitize(t.Title()))
		if i == m.active {
			labels[i] = theme.ActiveTab.Render(label)
		} else {
			labels[i] = theme.Muted.Render(label)
		}
	}
	return strings.Join(labels, " ")
}

func (m *Model) renderStatus() string {
	if m.cmdline != nil {
		return textutil.Fit(m.cmdline.View(m.width, 1), m.width)
	}
	parts := []string{theme.Mode.Render(" NORMAL ")}
	if len(m.pending) > 0 {
		parts = append(parts, m.keys.Display(m.pending))
	}
	if loc := m.location(); loc != "" {
		parts = append(parts, loc)
	}
	if m.status != "" {
		style := theme.Muted
		if m.statusErr {
			style = theme.Error
		}
		parts = append(parts, style.Render(textutil.Sanitize(m.status)))
	}
	left := strings.Join(parts, "  ")
	right := theme.Muted.Render("<space> leader · ? help · : command")
	gap := m.width - ansi.StringWidth(left) - ansi.StringWidth(right)
	if gap < 1 {
		return textutil.Fit(left, m.width)
	}
	return left + strings.Repeat(" ", gap) + right
}

func (m *Model) location() string {
	var conn string
	var ns db.Namespace
	if t := m.activeTab(); m.focus == focusTabs && t != nil {
		var ref db.TableRef
		conn, ref = t.Target()
		ns = ref.Namespace
	} else if n := m.tree.Selected(); n != nil {
		conn, ns = n.Conn, n.NS
	}
	var parts []string
	for _, p := range []string{conn, ns.Database, ns.Schema} {
		if p != "" {
			parts = append(parts, p)
		}
	}
	return strings.Join(parts, "/")
}
```

```go
// Command sqlvim is a terminal SQL client with vim keys.
package main

import (
	"flag"
	"fmt"
	"io"
	"log/slog"
	"os"
	"path/filepath"

	tea "charm.land/bubbletea/v2"

	"github.com/shepherd-yun/sqlvim/internal/config"
	"github.com/shepherd-yun/sqlvim/internal/db/mysql"
	"github.com/shepherd-yun/sqlvim/internal/db/postgres"
	"github.com/shepherd-yun/sqlvim/internal/secret"
	"github.com/shepherd-yun/sqlvim/internal/session"
	"github.com/shepherd-yun/sqlvim/internal/ui"
)

// version is set at build time via -ldflags "-X main.version=...".
var version = "dev"

func main() {
	if err := run(os.Args[1:], os.Stdout); err != nil {
		fmt.Fprintln(os.Stderr, "sqlvim:", err)
		os.Exit(1)
	}
}

func run(args []string, stdout io.Writer) error {
	fs := flag.NewFlagSet("sqlvim", flag.ContinueOnError)
	showVersion := fs.Bool("version", false, "print version and exit")
	debug := fs.Bool("debug", false, "write debug-level logs")
	configDir := fs.String("config", "", "config directory (default $XDG_CONFIG_HOME/sqlvim or ~/.config/sqlvim)")
	if err := fs.Parse(args); err != nil {
		return err
	}
	if *showVersion {
		fmt.Fprintln(stdout, "sqlvim", version)
		return nil
	}

	dir := *configDir
	if dir == "" {
		d, err := config.Dir()
		if err != nil {
			return err
		}
		dir = d
	}
	cfg, err := config.Load(dir)
	if err != nil {
		return err
	}
	if len(cfg.Connections) == 0 {
		return fmt.Errorf("no connections configured: create %s (see docs/examples/connections.toml)",
			filepath.Join(dir, "connections.toml"))
	}

	logPath, closeLog, err := setupLogging(*debug)
	if err != nil {
		return err
	}
	defer closeLog()

	mgr := session.NewManager(cfg.Connections, secret.Keyring{}, mysql.Driver{}, postgres.Driver{})
	defer mgr.CloseAll()

	slog.Info("starting", "version", version, "connections", len(cfg.Connections))
	if _, err := tea.NewProgram(ui.New(mgr, cfg.Settings)).Run(); err != nil {
		slog.Error("program exited", "err", err)
		return fmt.Errorf("%w (log: %s)", err, logPath)
	}
	return nil
}

func setupLogging(debug bool) (path string, closeFn func(), err error) {
	dir, err := config.StateDir()
	if err != nil {
		return "", nil, err
	}
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return "", nil, err
	}
	path = filepath.Join(dir, "sqlvim.log")
	f, err := os.OpenFile(path, os.O_CREATE|os.O_APPEND|os.O_WRONLY, 0o600)
	if err != nil {
		return "", nil, err
	}
	level := slog.LevelInfo
	if debug {
		level = slog.LevelDebug
	}
	slog.SetDefault(slog.New(slog.NewTextHandler(f, &slog.HandlerOptions{Level: level})))
	return path, func() { _ = f.Close() }, nil
}
```
